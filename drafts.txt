async function submission(form) {
  const submitBtns = [...form.querySelectorAll(`button[type=submit]`)];
  const intervalId = setInterval(() => {
    for (const submitBtn of submitBtns) {
      submitBtn.disabled = !form.checkValidity();
    }
  }, 1000);
  form.addEventListener("input", () => {
    for (const submitBtn of submitBtns) {
      submitBtn.disabled = !form.checkValidity();
    }
  });
  return await new Promise((resolve, reject) => {
    form.addEventListener("submit", (event) => {
      event.preventDefault();
      clearInterval(intervalId);
      form.reset();
      for (const submitBtn of form.querySelectorAll(`button[type=submit]`)) {
        submitBtn.disabled = !form.checkValidity();
      }
      resolve(new FormData(form, event.submitter));
    });
  });
}


/** @param {HTMLFormElement} form */
function bindSubmitComponent(form) {
  const update = () => {
    form.querySelectorAll(`button[type=submit]`).forEach(btn => {
      btn.disabled = !form.checkValidity();
    });
  };
  form.addEventListener("input", update);
  form.addEventListener("reset", () => setTimeout(update, 0));
}

async function submission(form) {
  return await new Promise((resolve) => {
    const onSubmit = (ev) => {
      ev.preventDefault();
      form.removeEventListener(onSubmit);
      resolve(new FormData(form, ev.submitter));
    };
    form.addEventListener("submit", onSubmit);
  });
